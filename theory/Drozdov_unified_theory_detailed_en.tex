\begin{abstract}
\noindent
This is an extended version of the preprint on a geometrodynamic model of
particles as defects of an elastic aether. Unlike the introductory exposition,
here \emph{every} structural statement is accompanied by a full derivation:
dimensional analysis, intermediate formulas, and, where appropriate, a proof in
the form of a theorem. The aim is to make all results reproducible and
checkable by the reader ``with a pencil''. The limits of the model (what is
derived, what is assumed, what is open) are traced through each proof, not only
in a final section.
\end{abstract}

\section{Notation and conventions}
The three-dimensional elastic continuum (the ``aether'') is described by a
displacement field $u_i(\mathbf{x},t)$. The dimensional basis is $(L,T,M)$.
Greek indices $a,b=1,2,3$ are internal (translational) indices of the gauge
group $T(3)$; Latin $i,j,k$ are spatial. Summation over repeated indices is
implied.

\begin{definition}[Basic elastic quantities]
Shear modulus $\mu$, $[ML^{-1}T^{-2}]$; density $\rho$, $[ML^{-3}]$;
transverse-wave speed $c=\sqrt{\mu/\rho}$, $[LT^{-1}]$; Burgers vector $b$,
$[L]$.
\end{definition}

\section{Gauge structure and the Maxwell dictionary}

\subsection{Distortion field and dislocation tensor}
We follow Lazar's notation \cite{lazar2010pla}; all indices are Latin,
$i,j,k,l=1,2,3$. Let $u_i$ be the displacement. Incompatibility is introduced by
the plastic distortion $\phi_{ij}$, the translational gauge field, whose first index
is translational. The elastic distortion is
\begin{equation}
\beta_{ij}=\partial_ju_i-\phi_{ij}.
\label{eq:beta}
\end{equation}
The dislocation-density tensor (torsion) is
\begin{equation}
T_{ijk}=\beta_{ik,j}-\beta_{ij,k}=-\left(\phi_{ik,j}-\phi_{ij,k}\right),
\label{eq:torsion}
\end{equation}
the second equality holding because $u_{i,kj}=u_{i,jk}$. Thus $T$ is determined by
the plastic part alone and is antisymmetric in its last pair of indices,
$T_{ijk}=-T_{ikj}$. The Burgers vector of a contour $\partial S$ follows from
Stokes' theorem ($\oint A_j\,dx_j=\int_S\epsilon_{klj}\partial_lA_j\,dS_k$):
\begin{equation}
b_i=\oint_{\partial S}\beta_{ij}\,dx_j=\int_S\alpha_{ik}\,dS_k,\qquad
\alpha_{ik}\equiv\epsilon_{klj}\beta_{ij,l}=\tfrac12\epsilon_{klj}T_{ilj}.
\label{eq:burgers}
\end{equation}
Here $\alpha_{ik}$ is Nye's tensor. The last equality holds because only the part of
$\epsilon_{klj}\beta_{ij,l}$ antisymmetric in $(l,j)$ survives, and
$\tfrac12(\beta_{ij,l}-\beta_{il,j})=\tfrac12T_{ilj}$. The sign of $b$ is a convention;
the classical literature \cite{hirth1982} often uses the opposite one.

\begin{lemma}[Dimension of the magnetic charge]
\label{lem:qm}
In the elastic basis the magnetic charge, defined as the flux of the dislocation
density $q_m=\int_S\alpha_{ik}\,dS_k$ \eqref{eq:burgers}, has the dimension of length.
\end{lemma}
\begin{proof}
The distortion $\beta_{ij}$ is dimensionless (a displacement gradient minus the
dimensionless $\phi$). From \eqref{eq:torsion}, $[T_{ijk}]=[\alpha_{ik}]=L^{-1}$, so
$[q_m]=L^{-1}\cdot L^{2}=L$. $\qed$
\end{proof}

\begin{remark}
Lemma~\ref{lem:qm} removes the apparent conflict with SI, where magnetic charge is
measured in $\text{A}\cdot\text{m}$: in a consistent elastic theory the ampere is
not a primary unit, and the dimension of the charge is dictated by the theory itself.
\end{remark}

\subsection{Dislocation current}
Let $v_i$ be the physical (in general incompatible) velocity of the medium. The
dislocation current is \cite[eq.~(1)]{lazar2010pla}
\begin{equation}
I_{ij}=-v_{i,j}+\dot\beta_{ij},\qquad [I_{ij}]=T^{-1}.
\label{eq:current}
\end{equation}
The density $T$ and the current $I$ are kinematically related by the Bianchi
identities derived below, \eqref{eq:bianchi}. In form these are the homogeneous
Maxwell equations, whence the dictionary $\mathbf B\leftrightarrow T$ (more precisely
$\alpha$), $\mathbf E\leftrightarrow I$ (Table~\ref{tab:corr}).

\subsection{Full derivation of the quasi-Maxwell system}
We give the derivation in detail, following Lazar's linear isotropic theory in
index notation \cite{lazar2010pla} (the differential-form version and the nonlinear
case are in \cite{lazar2010}). In this subsection all indices are Latin and spatial,
$i,j,k,l=1,2,3$; summation over repeated indices is implied.

\paragraph{Kinematics.} The basic fields are the elastic distortion $\beta_{ij}$
(dimensionless) and the physical velocity $v_i$ $[LT^{-1}]$; both are incompatible.
The dislocation density and the dislocation current are \cite[eq.~(1)]{lazar2010pla}
\begin{equation}
T_{ijk}=\beta_{ik,j}-\beta_{ij,k},\qquad I_{ij}=-v_{i,j}+\dot\beta_{ij},
\label{eq:TI_lazar}
\end{equation}
with $\beta_{ij,k}\equiv\partial_k\beta_{ij}$, $\dot\beta_{ij}\equiv\partial_t\beta_{ij}$;
$[T_{ijk}]=L^{-1}$, $[I_{ij}]=T^{-1}$. These are the same definitions as \eqref{eq:torsion} and \eqref{eq:current}; the
first index is translational.

\paragraph{Lagrangian.} The Lagrangian density has the bilinear form
\cite[eq.~(3)]{lazar2010pla}
\begin{equation}
\mathcal{L}=T-W=\tfrac12\,p_iv_i+\tfrac12\,D_{ij}I_{ij}-\tfrac12\,\sigma_{ij}\beta_{ij}
-\tfrac14\,H_{ijk}T_{ijk}.
\label{eq:lagr}
\end{equation}
\textit{Scalar character.} All indices are contracted in each term: one in the first,
two in the second and third, three in the fourth. No index is free, and $\mathcal L$ is
a scalar. Each term has the dimension of an energy density $[ML^{-1}T^{-2}]$:
$[p_iv_i]=\frac{M}{L^2T}\cdot\frac LT$, $[D_{ij}I_{ij}]=\frac M{LT}\cdot\frac1T$,
$[\sigma_{ij}\beta_{ij}]=\frac M{LT^2}$, $[H_{ijk}T_{ijk}]=\frac M{T^2}\cdot\frac1L$.

\paragraph{Responses.} The canonically conjugate quantities are
\cite[eq.~(4)]{lazar2010pla}
\begin{equation}
p_i=\frac{\partial\mathcal L}{\partial v_i},\quad
\sigma_{ij}=-\frac{\partial\mathcal L}{\partial\beta_{ij}},\quad
D_{ij}=\frac{\partial\mathcal L}{\partial I_{ij}},\quad
H_{ijk}=-2\frac{\partial\mathcal L}{\partial T_{ijk}}:
\label{eq:responses}
\end{equation}
the momentum, the force stress, the dislocation momentum flux and the pseudomoment
stress. The factor $2$ in $H$ compensates the coefficient $\tfrac14$ and the
antisymmetry $T_{ijk}=-T_{ikj}$.

\paragraph{Constitutive relations.} For a linear isotropic medium
\cite[eqs.~(10)--(13)]{lazar2010pla}
\begin{equation}
\begin{aligned}
p_i&=\rho\,v_i,\\
\sigma_{ij}&=\lambda\,\delta_{ij}\beta_{kk}+\mu(\beta_{ij}+\beta_{ji})+\gamma(\beta_{ij}-\beta_{ji}),\\
D_{ij}&=d_1\,\delta_{ij}I_{kk}+d_2(I_{ij}+I_{ji})+d_3(I_{ij}-I_{ji}),\\
H_{ijk}&=c_1\,T_{ijk}+c_2(T_{jki}-T_{kji})+c_3(\delta_{ij}T_{llk}-\delta_{ik}T_{llj}).
\end{aligned}
\label{eq:constit}
\end{equation}
These are ten constants: $\rho$ and nine moduli, $[\lambda,\mu,\gamma]=ML^{-1}T^{-2}$,
$[d_{1,2,3}]=ML^{-1}$, $[c_{1,2,3}]=MLT^{-2}$. Non-negativity of the energy
$T+W\ge0$ requires \cite[eq.~(14)]{lazar2010pla}: $\rho,\mu,\gamma\ge0$,
$3\lambda+2\mu\ge0$; $d_2,d_3\ge0$, $3d_1+2d_2\ge0$; $c_1-c_2\ge0$,
$c_1+2c_2\ge0$, $c_1-c_2+2c_3\ge0$.

\textit{Relation to the elasticity tensor.} The elastic part is
$W_{\text{el}}=\tfrac12\sigma_{ij}\beta_{ij}=\tfrac12 C_{ijkl}\beta_{ij}\beta_{kl}$ with
\[
C_{ijkl}=\lambda\,\delta_{ij}\delta_{kl}+(\mu+\gamma)\,\delta_{ik}\delta_{jl}+(\mu-\gamma)\,\delta_{il}\delta_{jk}.
\]
Each of the four indices of $C$ is contracted with an index of one of the two
distortions. For $\gamma=0$ this is the usual isotropic Lam\'e tensor. The constant
$\gamma$ assigns energy to the antisymmetric (rotational) part $\beta_{[ij]}$; it is
absent in classical elasticity, where the distortion is a displacement gradient and
a rotation carries no energy.

\textit{Remark on the earlier form.} In the previous version of this paper the elastic
term was written as $\tfrac12C_{ijkl}\beta^i_{\ a}\beta^{ja}$. There the indices $k,l$
are not contracted, so the expression is not a scalar; moreover,
$D_{ab}I^a_iI^{bi}$ and $A_{ab}T^a_{ij}T^{bij}$ had one constant each instead of three.
The present form \eqref{eq:lagr}--\eqref{eq:constit} corrects both points. The earlier
simplified form is the special case $d_1=0$, $d_2=d_3$ and $c_2=c_3=0$. Since the
results of the following sections use only the structure of the equations and not
the specific values of $d_i,c_i$, they are not affected by this correction.

\paragraph{Equations of motion.} The Euler--Lagrange equations with respect to $v_i$
and $\beta_{ij}$ \cite[eqs.~(5)--(6)]{lazar2010pla}, with external sources $p^0_i$,
$\sigma^0_{ij}$ (null Lagrangian $\sigma^0_{ij}\beta_{ij}-p^0_iv_i$, with
$\dot p^0_i-\sigma^0_{ij,j}=0$), take in terms of the responses \eqref{eq:responses}
the form \cite[eqs.~(7)--(8)]{lazar2010pla}
\begin{align}
D_{ij,j}+p_i&=p^0_i, &&\text{(dislocation momentum balance; Gauss-law analogue)}\label{eq:gauss}\\
\dot D_{ij}+H_{ijk,k}+\sigma_{ij}&=\sigma^0_{ij}. &&\text{(stress balance; Amp\`ere--Maxwell analogue)}\label{eq:ampere}
\end{align}
Differentiating \eqref{eq:ampere} with respect to $x_j$ and \eqref{eq:gauss} with
respect to $t$ and subtracting gives the integrability condition
$\dot p_i-\sigma_{ij,j}=0$, the ordinary equation of motion of the elastic medium
\cite[eq.~(9)]{lazar2010pla}. The terms $D_{ij,j}$ and $H_{ijk,k}$ cancel by symmetry.

\textit{How this pair corresponds to Maxwell's equations.} In \eqref{eq:ampere} the term
$\dot D$ corresponds to the displacement current $-\frac1c\dot{\mathbf D}$, and
$H_{ijk,k}$ to the curl $\nabla\times\mathbf H$ (the divergence of a tensor
antisymmetric in $jk$ is equivalent to the curl of the vector
$\tfrac12\epsilon_{jkl}H_{ijk}$). The stress $\sigma_{ij}$ plays the role of the
current, and $p_i$ in \eqref{eq:gauss} the role of the charge density. Thus the
sources of the dislocation field are the momentum and the stresses of the elastic
medium.

\paragraph{Bianchi identities.} The definitions \eqref{eq:TI_lazar} imply
identically \cite[eq.~(2)]{lazar2010pla}
\begin{equation}
\epsilon_{jkl}\,T_{ijk,l}=0,\qquad \dot T_{ijk}+I_{ij,k}-I_{ik,j}=0.
\label{eq:bianchi}
\end{equation}
The first is the analogue of $\nabla\cdot\mathbf B=0$ (no sources of dislocation
density: a dislocation line cannot end inside the medium). The second is the analogue
of Faraday's law. Check of the second: $\dot T_{ijk}=\dot\beta_{ik,j}-\dot\beta_{ij,k}$
and $I_{ij,k}-I_{ik,j}=-v_{i,jk}+\dot\beta_{ij,k}+v_{i,kj}-\dot\beta_{ik,j}$; the sum
vanishes because $v_{i,jk}=v_{i,kj}$.
Comparing \eqref{eq:constit}--\eqref{eq:bianchi} with Maxwell's equations gives
the dictionary of Table~\ref{tab:corr} (below): the strengths $T,I$ correspond to
$\mathbf B,\mathbf E$, and the responses $H,D$ to $\mathbf H,\mathbf D$.

\paragraph{Peach--Koehler force.} In index notation the force density acting on the
dislocations is
\begin{equation}
f_k=\sigma_{ij}T_{ijk}-p_iI_{ik},
\label{eq:pk}
\end{equation}
which is the component form of eq.~(31) of \cite{lazar2010},
$f^a=(e_a\lrcorner I^b)p_b+(e_a\lrcorner T^b)\wedge\Sigma_b$. The derivation of
\eqref{eq:pk} is ours and is not taken from \cite{lazar2010pla}. Substituting
\eqref{eq:torsion} and \eqref{eq:current}:
\[
f_k=\sigma_{ij}\beta_{ik,j}-p_i\dot\beta_{ik}-\sigma_{ij}\beta_{ij,k}+p_iv_{i,k}.
\]
Using the elastic equation of motion $\dot p_i=\sigma_{ij,j}$ (the integrability
condition of \eqref{eq:gauss}--\eqref{eq:ampere}), the first two terms combine:
$\sigma_{ij}\beta_{ik,j}-p_i\dot\beta_{ik}=\partial_j(\sigma_{ij}\beta_{ik})-\partial_t(p_i\beta_{ik})-\beta_{ik}(\sigma_{ij,j}-\dot p_i)=\partial_j(\sigma_{ij}\beta_{ik})-\partial_t(p_i\beta_{ik})$.
For a homogeneous linear medium $C_{ijlm}=C_{lmij}$, so
$\sigma_{ij}\beta_{ij,k}=\partial_kW$, and $p_iv_{i,k}=\rho v_iv_{i,k}=\partial_kT$.
Hence
\begin{equation}
f_k=-\partial_t\!\left(p_i\beta_{ik}\right)+\partial_j\!\left(\sigma_{ij}\beta_{ik}-\delta_{jk}(W-T)\right).
\label{eq:pk_balance}
\end{equation}
The force density is the divergence of a four-dimensional Eshelby-type tensor.
Therefore $f_k$ is the configurational force on the defects, and $\int f_k\,dV$ over a
region containing a defect reduces to a flux through its boundary. Here $W-T=-\mathcal L$ appears, i.e.
\eqref{eq:pk_balance} is the Eshelby tensor $\mathcal L\delta_{jk}+\sigma_{ij}\beta_{ik}$
of dynamic elasticity. The minus sign in front of $p_iI_{ik}$ in \eqref{eq:pk} is fixed
precisely by requiring this identity. We checked it numerically on random smooth
fields: with this sign the residual $-\beta_{ik}(\sigma_{ij,j}-\dot p_i)$ vanishes on
the equations of motion, and with a plus sign it does not. The sign has not been
checked against an index form of Lazar's eq.~(31); in \cite{lazar2010} it depends on
his conventions for forms.

\textit{Straight-dislocation limit.} For a dislocation line with tangent $\xi_l$ and
Burgers vector $b_i$ we have $\alpha_{ik}=b_i\xi_k\delta^{(2)}$, and inverting
\eqref{eq:burgers}, $T_{ijk}=\epsilon_{jkl}\alpha_{il}$. For a line at rest ($I=0$)
the force per unit length is
$f_k=\sigma_{ij}b_i\epsilon_{jkl}\xi_l=[\boldsymbol\xi\times(\sigma\mathbf b)]_k$.
This is the classical Peach--Koehler force $(\sigma\mathbf b)\times\boldsymbol\xi$
with the opposite sign, which reflects the sign convention of \eqref{eq:burgers}.

\textit{Analogy with the Lorentz force.} The term $-p_iI_{ik}$ corresponds to
$\rho\mathbf E$ (up to the sign convention) ($p\leftrightarrow\rho$, $I\leftrightarrow\mathbf E$), and the term
$\sigma_{ij}T_{ijk}$ to $\frac1c\mathbf j\times\mathbf B$ ($\sigma\leftrightarrow\mathbf j$,
$T\leftrightarrow\mathbf B$). This is the formal basis of the whole analogy: the
electrodynamics of the particle-defect is the dynamics of its dislocation field.

\section{The particle: a precessing dislocation dipole}

\subsection{Kinematics of the special point}
A ring of radius $a$ lies in a fixed plane; the special point (a localized
charge $q_m=\Delta b$) runs along it: azimuth $\varphi(t)=\varphi_0+\omega t$.
Position, velocity and acceleration:
\begin{align}
\mathbf{r}(t) &= a(\cos\varphi,\ \sin\varphi,\ 0),\\
\mathbf{v}(t) &= a\omega(-\sin\varphi,\ \cos\varphi,\ 0),\quad |\mathbf{v}|=a\omega,\\
\mathbf{W}(t) &= -a\omega^2(\cos\varphi,\ \sin\varphi,\ 0),\quad |\mathbf{W}|=a\omega^2.
\label{eq:accel}
\end{align}

\subsection{Mass as elastic self-interaction}
\begin{proposition}[Dimensional closure of mass]
The self-energy of a dislocation ring $E=\tfrac12\mu b^2 a\,\ln(8a/r_c)$
(Hirth--Lothe \cite{hirth1982}), after dropping the logarithm, yields the mass
$m=\mu b^2 a/c^2$ with the correct dimension $[M]$.
\end{proposition}
\begin{proof}
$[\mu b^2 a]=(ML^{-1}T^{-2})(L^2)(L)=ML^{2}T^{-2}=[\text{energy}]$. Division by
$c^2$ $[L^2T^{-2}]$ gives $[M]$. $\qed$
\end{proof}

\subsection{Dipole from a pair of rings (antiphase)}
Two rings at $z=\pm d/2$, special points in antiphase
($\varphi_{\text{low}}=\varphi_{\text{up}}+\pi$), charges $\pm q_m$. The
instantaneous dipole moment $\mathbf{p}=q_m\mathbf{r}_+ - q_m\mathbf{r}_-$.
A direct computation:
\begin{equation}
p_z = q_m\cdot d = \text{const},\qquad
|\mathbf{p}_{xy}| = 2 q_m a\ \ (\text{rotates with }\omega).
\end{equation}
The axial component is constant (a static dipole), the transverse one precesses.

\subsection{Stability: the centrifugal-barrier theorem}
\begin{theorem}[Equilibrium of the pair]
\label{th:stab}
Let the pair be subject to attraction $U_{\text{att}}=-A/d$ (opposite charges)
and let the angular momentum $J=m a^2\omega$ be conserved. Then the effective
energy $E(d)=-A/d+J^2/(2md^2)$ has a unique minimum at
$d_{\text{eq}}=J^2/(mA)$, and as $\omega\to0$ (i.e.\ $J\to0$) the equilibrium
vanishes ($d_{\text{eq}}\to0$).
\end{theorem}
\begin{proof}
$E'(d)=A/d^2-J^2/(md^3)=0 \Rightarrow d_{\text{eq}}=J^2/(mA)$. The second
derivative $E''(d_{\text{eq}})=-2A/d^3+3J^2/(md^4)$; substituting $d_{\text{eq}}$
gives $E''=A^4 m^3/J^6>0$ --- a minimum. As $J\to0$ the barrier term vanishes,
only attraction remains, and the minimum shifts to $d\to0$ (collapse). $\qed$
\end{proof}

\begin{remark}[Which coordinate is stabilized (correction)]
\label{rem:stab_axis}
The centrifugal term $J^2/(2md^2)$ refers to the \emph{radial} coordinate in the
plane of rotation: it arises from $J=m\rho^2\omega$ with $\rho$ the distance to
the axis. The axial distance $d$ between the ring planes does not enter $J$, so
Theorem~\ref{th:stab} applies literally only if $d$ is read as a radial distance,
i.e.\ the orbit radius of the special point. In the previous version of this paper
$d_{\text{eq}}$ was identified with the axial gap of the pair, which was incorrect.
Check: for $J=\hbar/2$, $A=e^2$ the theorem gives
$d_{\text{eq}}=\lambdabar_C/(4\alpha)=a_0/4$, an atomic scale rather than the size
of the particle. The axial gap $d$ is set by the elastic interaction of the two
rings and is not computed in the present work
(see Section~\ref{sec:spin_subluminal}).
\end{remark}

\begin{remark}[Annihilation]
Collapse upon loss of precession is Frenkel recombination: an interstitial loop
($+\Delta b$) and a vacancy loop ($-\Delta b$) give a total Burgers vector
$\mathbf{b}_++\mathbf{b}_-=0$ --- the perfect lattice. Thus precession (spin) is
the mechanism protecting the particle--antiparticle pair from self-annihilation.
\end{remark}

\subsection{Spin and statistics: the configuration period}
\begin{proposition}[Period of the pair and statistics]
For a pair of ordinary rings the configuration returns at
$\varphi\to\varphi+2\pi$ (a boson). For a pair of M\"obius rings, where the sign
of the Burgers vector is modulated as $\cos(\varphi/2)$, the sign is restored
only at $\varphi\to\varphi+4\pi$ (a fermion, spin $1/2$).
\end{proposition}
\begin{proof}
The twist $\cos(\varphi/2)$ is antiperiodic under $2\pi$:
$\cos((\varphi+2\pi)/2)=\cos(\varphi/2+\pi)=-\cos(\varphi/2)$, and periodic
under $4\pi$. Hence a spinor (double-valued) structure arises precisely for the
M\"obius ring. $\qed$
\end{proof}

\begin{lemma}[Frank vector of the $\cos(\varphi/2)$ twist]
\label{lem:franck}
The Burgers twist $b(\varphi)=\cos(\varphi/2)$ is equivalent to a Frank
rotation by the fixed angle $\Omega=\pi$ per circuit $\varphi:0\to2\pi$, i.e.\ a
wedge \emph{half-disclination}. The Frank index $\nu=\Omega/2\pi=1/2$, giving
spin $1/2$.
\end{lemma}
\begin{proof}
The twist phase is $\varphi_{\text{tw}}(\varphi)=\varphi/2$. The total rotation
of the Burgers direction per circuit is
$\Omega=\oint d\varphi_{\text{tw}}=\int_0^{2\pi}\tfrac12\,d\varphi=\pi$. Over
$4\pi$ the rotation is $2\pi$ (full return), reproducing the spinor
double-valuedness of the period proposition. Through the Osipov spin connection
(\texttt{arXiv:1001.4450}) $\Omega_\varphi=i\omega\sigma_3$ with flux
$\oint\Omega_\varphi\,d\varphi=2\pi\nu$ we get $\nu=1/2$ and the magnetic moment
$\mu=\nu=1/2$ as the flux of the spin connection (not a loop current). $\qed$
\end{proof}

\begin{remark}[M\"obius dipole and the link to the $SO(3)$ sector]
The electron in the model is a pair of M\"obius rings with \emph{opposite}
directions of the Burgers vector: one ring carries the jump $+\Delta b$
(pole~$N$), the other $-\Delta b$ (pole~$S$); together they give a static
magnetic dipole $q_m d$. This distinguishes the electron (M\"obius pair, period
$4\pi$, fermion) from a boson (pair of ordinary rings, period $2\pi$).
Crucially, the Burgers twist $b(\varphi)=\cos(\varphi/2)$ is a Frank rotation by
$\pi$ per circuit --- a \emph{disclination} (rotational, $SO(3)$) degree of
freedom. Thus the M\"obius sector already lies in $SO(3)$: the magnetic moment
$\mu=\nu=n/2$ is derived as the flux of the spin connection
$\oint\omega_\varphi\,d\varphi=2\pi\nu$ (Osipov, \texttt{arXiv:1001.4450}),
not as a loop current. Hydrodynamically the same object is a vortex: the
vorticity $\nabla\times v$ corresponds to the Frank density, the circulation to
the disclination strength. The precession of the special point ($\omega>0$),
required for the stability of the pair, is that vortex; a particle without
rotation collapses.
\end{remark}

\section{The magnetic moment: why a dipole, not a current}

\begin{theorem}[Vanishing of the current axial moment]
\label{th:mu0}
For a pair of rings in antiphase with opposite charges, the period-averaged
axial ``loop-current'' magnetic moment $\boldsymbol{\mu}_{\text{cur}}=
\tfrac12\oint q\,\mathbf{r}\times\mathbf{v}$ is zero.
\end{theorem}
\begin{proof}
Both rings run in the same sense, so $\mathbf{r}\times\mathbf{v}$ is the same
for both and points along $+z$: $(\mathbf{r}\times\mathbf{v})_z=a^2\omega$. But
the charges are opposite: $q_{\text{up}}=+q_m$, $q_{\text{low}}=-q_m$. The
contributions $\tfrac12 q_m a^2\omega$ and $-\tfrac12 q_m a^2\omega$ cancel:
$\mu_{\text{cur},z}=0$. $\qed$
\end{proof}

\begin{remark}
Theorem~\ref{th:mu0} means that the electron's magnetic moment in the model is
a \emph{static} dipole moment $\mu_{\text{mag}}=q_m d$, not a loop-current
moment. This is consistent with the ``current'' along the ring being magnetic,
so that applying the electric loop-current formula to it is incorrect.
\end{remark}

\begin{proposition}[Dimensional failure of the $g$-factor from the ED formula]
Attempting to define $g$ via $\mu_{\text{mag}}=g(q_m/2m)S$ in the elastic basis
gives $g$ the dimension $T/L$, not a dimensionless quantity; hence the
electrodynamic formula does not transfer literally, and the classical $g=2$ is
not obtained in the ring model (one gets $g=1$).
\end{proposition}
\begin{proof}
$[\mu_{\text{mag}}]=[q_m][d]=L^2$. $[S]=[m a^2\omega]=M L^2 T^{-1}$.
$[(q_m/m)S]=(L/M)(ML^2T^{-1})=L^3T^{-1}$. Then
$[g]=[\mu_{\text{mag}}]/[(q_m/m)S]=L^2/(L^3T^{-1})=T/L\neq 1$. $\qed$
\end{proof}


\noindent\textit{Remark.} This statement concerns expressing $g$ through $q_m$ in elastic units. In Section~\ref{sec:spin_subluminal} $g$ is defined through the observed charge $e$, and no dimensional difficulty arises.

\section{A subluminal classical model of spin and the gyromagnetic ratio}
\label{sec:spin_subluminal}

\paragraph{The difficulty in classical electrodynamics.} If the electron is a
charged sphere of radius $R=r_e/2$ (field energy $e^2/2R=m_ec^2$) and the Bohr
magneton is produced by its rotation, $\mu_B=e\omega R^2/3c$, the equatorial speed is
\begin{equation}
\frac{v}{c}=\frac{3\mu_B}{eR}=\frac{3}{\alpha}\approx411 .
\label{eq:v_classical}
\end{equation}
This is not an accidental number but an exact factor $1/\alpha$. It arises because
the magnetic moment $\mu$ and the mechanical moment $S$ are produced by \emph{the same}
moving charge and therefore sit on the same lever arm. Hence $\mu/S=e/(2mc)$, and
the value $\mu_B$ at a radius of order $r_e$ requires superluminal rotation (Abraham;
the difficulty was noted by Uhlenbeck and Goudsmit in 1925).

\paragraph{Separate carriers in the model.} In the ring pair the two moments are of
different nature.
\begin{enumerate}[nosep]
\item The \emph{mechanical moment} is produced by the motion of the special point
along a ring of radius $a$: $S=mav$. Its value $S=\hbar/2$ is fixed by the topology of
the twist, Frank vector $\Omega=\pi$, $\nu=\tfrac12$ (Lemma~\ref{lem:franck}), not
by fitting a speed.
\item The \emph{magnetic moment} is the static dipole of the two Burgers jumps
$\pm\Delta b=\pm q_m$ separated by the axial gap $d$: $\mu=q_md$. No rotation is
needed for it; the current contribution of the pair vanishes
(Theorem~\ref{th:mu0}).
\end{enumerate}

\begin{proposition}[Subluminal spin]
\label{prop:subluminal}
For $a=\lambdabar_C$ and $S=\hbar/2$ the speed of the special point is $v=c/2$.
\end{proposition}
\begin{proof}
$v=S/(ma)=(\hbar/2)/(m\lambdabar_C)=(\hbar/2)\,c/\hbar=c/2$. $\qed$
\end{proof}
The factor $1/\alpha$ of \eqref{eq:v_classical} does not appear in the model: $S$ sits
on the arm $a=\lambdabar_C$, not on $r_e=\alpha\lambdabar_C$. Were $S$ also carried on
$r_e$, one would get $v=c/(2\alpha)\approx68c$, the same difficulty.
\emph{Status: proved} under the calibration $a=\lambdabar_C$; the choice of $a$ is an
assumed calibration, not a derivation.

\paragraph{The gyromagnetic ratio.} Compare $\mu=q_md$ with the observed moment. In
the Coulomb channel of the model the pole strength $q_m$ appears as the observed
charge $e$ (in the Gaussian calibration $q_m\to e$). Then
\begin{equation}
g\equiv\frac{\mu}{(e/2mc)\,S}=\frac{e\,d}{e\hbar/(4mc)}=\frac{4d}{\lambdabar_C},
\qquad
g=2\iff d=\tfrac12\lambdabar_C=\tfrac12a .
\label{eq:g_geometry}
\end{equation}
Thus in the model $g$ is not postulated but reduces to a single geometric ratio: the
axial gap of the pair relative to its radius. The classical ``rotating charge''
formula gives $g=1$ because there $d$ and $a$ coincide with one radius. In the model
they are independent.

\paragraph{What is not proved.}
\begin{enumerate}[nosep]
\item \emph{The value $d/a=\tfrac12$ is not derived.} It requires the elastic
equilibrium of two coaxial rings of opposite sign in the axial coordinate: the
attraction of the poles against the elastic repulsion of the rings with the moduli
\eqref{eq:constit}. The centrifugal Theorem~\ref{th:stab} does not give this quantity
(Remark~\ref{rem:stab_axis}). We also checked the radial equilibrium of a single
ring: the line tension $T$ against the centrifugal term $S^2/(2ma^2)$ with
$m=\kappa Ta/c^2$ gives $a/\lambdabar_C=\tfrac12\sqrt{3\kappa/4\pi}$. It depends on
the unknown coefficient $\kappa$ and fixes neither $a=\lambdabar_C$ nor $d$.
\item \emph{Identifying $\mu=q_md$ with the measured moment.} If magnetic charges
are primary, the observed ``magnetic moment'' is the quantity dual to an electric
dipole moment. The substitution $q_m\to e$ in \eqref{eq:g_geometry} is a duality
convention that needs separate justification.
\item The \emph{anomalous part} $g-2=\alpha/\pi+\dots$ is not addressed in the model.
\item \emph{Predecessor model.} Kochetov and Osipov \cite{osipov2010} studied Dirac
fermions on a disclination: the spin connection $\Omega_\varphi=i\omega\sigma_3$ with
flux $2\pi\nu$ gives $J=\tfrac12$, and $g=2$ follows from the spinor structure (the
Pauli identity). Our model replaces the point-particle spinor by a classical ring
pair. If the classical condition $d=a/2$ is derived, it should reproduce their result.
\end{enumerate}
Result: the model gives a \emph{subluminal} classical picture of spin (proved) and
reduces $g=2$ to the condition $d=a/2$ on the geometry of the pair (formulation).
Deriving this condition from elastic equilibrium is an open problem.

\section{The Fermi film and gravity: full derivation}

\subsection{Setup}
The Fermi film is the three-dimensional boundary $\partial\Omega$ of the
four-dimensional action integral, stretched orthogonally to the world tube. Its
contribution to the action:
\begin{equation}
S_{\partial} = \gamma\int_{\partial\Omega} d\Sigma,
\end{equation}
where $\gamma$ is the film elastic modulus $[ML^{-1}T^{-2}]=[\text{Pa}]$.

\subsection{Source of deformation}
\begin{proposition}[Positivity of the source]
The period-averaged source of film deformation from a precessing charge
\begin{equation}
\sigma = \mu b^2 \frac{\langle W^2\rangle}{c^2}
= \mu b^2 \frac{\omega^4 a^2}{c^2}
\end{equation}
is strictly positive and independent of the sign of the charge.
\end{proposition}
\begin{proof}
From \eqref{eq:accel}, $|\mathbf{W}|=a\omega^2$, hence $\langle W^2\rangle=
a^2\omega^4>0$ (average of a constant). The sign of the charge enters through
$b^2$, i.e.\ drops out. $\qed$
\end{proof}

\subsection{The inverse-square law}
\begin{theorem}[$1/R^2$ as a consequence of the Green function]
\label{th:invsq}
If the film deformation obeys the equilibrium equation of an elastic membrane
\begin{equation}
\gamma\,\nabla^2\Delta(\mathbf{r})=-\sigma\,\delta^3(\mathbf{r}-\mathbf{r}_0),
\label{eq:poisson}
\end{equation}
then the interaction potential of two sources decays as $1/R$, and the force as
$1/R^2$.
\end{theorem}
\begin{proof}
Equation \eqref{eq:poisson} is the Poisson equation. Its fundamental solution
(the Laplacian Green function in $\mathbb{R}^3$) is $G(\mathbf{r})=1/(4\pi|\mathbf{r}|)$,
whence
\begin{equation}
\Delta(\mathbf{r})=\frac{\sigma}{4\pi\gamma\,|\mathbf{r}-\mathbf{r}_0|}.
\end{equation}
The energy of a second source $\sigma_2$ in the field of the first:
$U(R)=\sigma_2\Delta(R)=\sigma_1\sigma_2/(4\pi\gamma R)\propto R^{-1}$.
The force $F=-dU/dR=\sigma_1\sigma_2/(4\pi\gamma R^2)\propto R^{-2}$.
The exponent $-2$ is fixed by the dimension of space ($n=3$: $G\sim R^{-(n-2)}$),
not by choice. $\qed$
\end{proof}

\subsection{Two stiffness scales}
\begin{proposition}[Impossibility of a single stiffness]
The modulus reproducing the electron mass and the Planck modulus of gravity are
incompatible: their ratio is $\sim 10^{84}$.
\end{proposition}
\begin{proof}
From $m_e c^2=\mu_{\text{aeth}}\,r_e^2\lambda_C$ (with $b=r_e$, $a=\lambda_C$):
$\mu_{\text{aeth}}=m_ec^2/(r_e^2\lambda_C)\approx2.7\times10^{28}$~Pa.
The Planck modulus $\gamma_P=c^7/(8\pi\hbar G^2)\approx1.8\times10^{112}$~Pa.
The ratio $\gamma_P/\mu_{\text{aeth}}\approx6.9\times10^{83}$. $\qed$
\end{proof}

\begin{remark}[Physical meaning of the moduli]
The quantity $\mu_{\text{aeth}}$ is the \emph{shear modulus of the vacuum}: the
stiffness of the aether with respect to shear deformation, of pressure
dimension $[\text{Pa}]$. It is through it that, in the $(L,T,M)$ basis (without
the ampere), the transverse-wave speed $c=\sqrt{\mu_{\text{aeth}}/\rho}$, the
particle mass $m=\mu_{\text{aeth}}b^2a/c^2$ and the Coulomb force
$F=\mu_{\text{aeth}}b^2/R\cdot(\text{geometry})$ are defined. Physically it
governs shear deformations (dislocations), from which charge and mass are
built, and sets the physics at the particle scale (of order the Compton length
$\lambda_C$). Its value $2.7\times10^{28}$~Pa is obtained by \emph{calibration}
to the electron (substituting $m_e$ and $r_e$ into the self-energy of the
dislocation ring), not derived from first principles.

The quantity $\gamma_P=c^7/(8\pi\hbar G^2)$ is the \emph{Planck stiffness of the
vacuum}, a second elastic modulus of the same dimension $[\text{Pa}]$,
responsible for the gravitational response (deformation of the Fermi film) at
the Planck scale $\ell_P=\sqrt{\hbar G/c^3}$. The gravitational constant $G$
enters it as an \emph{input} parameter. The colossal ratio
$\gamma_P/\mu_{\text{aeth}}\sim10^{84}$ is a direct expression of the weakness
of gravity and is the main unsolved problem of the model (the origin of the
hierarchy of the two scales).
\end{remark}

\begin{remark}
Thus the gravitational constant $G$ entering $\gamma_P$ is an \emph{input}
parameter of the aether stiffness. We do not claim to derive it from $(\hbar,c)$;
only the \emph{structure} is proved (Theorem~\ref{th:invsq}), not the numerical
value. This is a fundamental difference from the fabricated versions, in which
$G$ was hidden inside the stiffness and ``extracted'' back (a tautology).
\end{remark}

\begin{remark}[Physical meaning of $\gamma_P$ and the nature of the hierarchy]
The Planck modulus is the Planck \emph{energy density} divided by $8\pi$:
\begin{equation}
\gamma_P=\frac{c^7}{8\pi\hbar G^2}=\frac{1}{8\pi}\,\frac{m_P c^2}{\ell_P^3},
\qquad \ell_P=\sqrt{\hbar G/c^3},\quad m_P=\sqrt{\hbar c/G}
\end{equation}
(numerical agreement $<1\%$). This is not an arbitrary combination but the
stiffness of the vacuum at the Planck scale; the factor $8\pi$ is the same as
in the Einstein equations. Likewise $\mu_{\text{aeth}}=m_e c^2/(r_e^2\lambda_C)$
is the energy density at the electron scale. The ratio of the two moduli
\begin{equation}
\frac{\gamma_P}{\mu_{\text{aeth}}}\sim\frac{m_P}{m_e}\ \text{(with a volume factor)}\sim10^{84}
\end{equation}
reduces to the ratio of the Planck and Compton scales, i.e.\ to the smallness
of $m_e/m_P$. Thus the hierarchy of the two stiffnesses is \emph{equivalent} to
the standard \emph{hierarchy problem} ($m_e\ll m_P$): the model does not solve
it, but neither does it introduce a new arbitrary parameter --- it inherits the
well-known smallness of $m_e/m_P$.
\end{remark}

\section{Six channels: a single table with derived laws}
\label{sec:channels}
Each decay law is a consequence of the proved structure:
\begin{itemize}[nosep]
\item \textbf{Coulomb} $1/R^2$: product of Burgers charges, the direct Coulomb
kernel (Theorem~\ref{th:invsq} for the field $T_{ijk}$).
\item \textbf{Magnetic} $1/R^4$: interaction of two static dipoles $q_m d$
(derivative of the dipole--dipole potential $\sim1/R^3$).
\item \textbf{Gravity} $1/R^2$: Theorem~\ref{th:invsq} for the source $\sigma>0$.
\item \textbf{Exchange} $1/R^3$: phase dependence $\cos2\Delta\varphi$, short-range.
\item \textbf{Weak} $e^{-R/R_W}$: massive mediator, twist chirality (parity violation).
\item \textbf{Strong} $\propto R$: fractional Burgers vector $b/3$ and a stacking-fault
ribbon (linear potential, confinement).
\end{itemize}
Two dimensionless numbers close the system:
\begin{equation}
\alpha=\frac{ke^2}{\hbar c}=\frac{1}{137.04},\qquad
\alpha_G=\Big(\frac{m_e}{m_{\text{Planck}}}\Big)^2=1.75\times10^{-45},\qquad
\frac{\alpha_G}{\alpha}=2.4\times10^{-43}.
\end{equation}

\section{Falsifiability}
The model gives testable predictions. For magneto-gravitational devices
(Roshchin--Godin type) the relative correction to weight from precession is
estimated as
\begin{equation}
\frac{\Delta G}{G}\sim\frac{W R}{c^2}\sim 10^{-13},
\end{equation}
which is $\sim\!12$ orders of magnitude smaller than the claimed effects
($\sim\!35\%$); the latter, if real, would \emph{refute} the model, not confirm
it. This is a sign of a scientific, not a fitted, theory.

\section{Limits: proved / assumed / open}
\paragraph{Proved (with the derivation above).} The dimension of the charge
(Lemma~\ref{lem:qm}); the closure of mass; stability and annihilation
(Theorem~\ref{th:stab}); statistics from the period; the vanishing of the
current moment (Theorem~\ref{th:mu0}); the positivity of the gravity source;
the $1/R^2$ law (Theorem~\ref{th:invsq}); the necessity of two scales; the
Frank vector $\Omega=\pi$ of the twist (Lemma~\ref{lem:franck}).

\paragraph{Assumed.} The values $\alpha$, $\alpha_G$; $G$ (embedded in
$\gamma_P$); the two stiffness scales.

\paragraph{Open.} The numerical values of $\alpha,\alpha_G$; the derivation of the condition $d=a/2$ that gives $g=2$
(Section~\ref{sec:spin_subluminal}; subluminal spin is proved, but the pair geometry
that fixes $g$ has not yet been derived from elastic equilibrium; the predecessor is
the Kochetov--Osipov spinor model of a disclination \cite{osipov2010}, where $g=2$
arises at the level of the Dirac equation); the
dynamical film stiffness (the dimensional factor in the source); the weak- and
strong-sector couplings.

\paragraph{The tensor ($J{=}2$) gravity channel --- a diagnosis.} The source
$\sigma\propto\langle W^2\rangle$ used above is the \emph{trace} (scalar, $J=0$)
of the quadrupole tensor $\langle W_iW_j\rangle$. The true tensor graviton
(spin-2 in Hehl's sense \cite{hehl1995}) lives in the trace-free part
$Q_{ij}=\langle W_iW_j\rangle-\tfrac13\delta_{ij}\langle W^2\rangle$. For a
single precessing ring with axis $\mathbf n$ a direct computation gives
$\langle W_iW_j\rangle=\tfrac12\omega^4a^2(\delta_{ij}-n_in_j)$, i.e.\
$Q_{ij}\neq0$ (an oblate quadrupole in the precession plane, eigenvalues
$\{-\tfrac13,+\tfrac16,+\tfrac16\}\omega^4a^2$). However, averaging over
isotropically oriented axes sends $\langle Q_{ij}\rangle\to0$: for an
unpolarized ensemble only the scalar $\langle W^2\rangle$ survives. Thus the
``scalarity'' of the model's gravity is a consequence of orientational
averaging, not of the absence of $J=2$. The tensor channel would manifest only
for \emph{polarized} (spin-aligned) matter. This is a diagnosis and a testable
prediction, not a derivation of the tensor graviton: the explicit field
equation for $Q_{ij}$ remains open.

\section{Conclusion}
A full derivation of all structural results of the model has been given. The
proofs are elementary (dimensional analysis, the Green function, the virial
theorem), but it was precisely their absence in earlier versions that led to
fitting. The proved, assumed and open parts are honestly separated. The next
steps are a rigorous spinor derivation of $g=2$ and this English version for
archival publication.
